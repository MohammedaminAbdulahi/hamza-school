   111→   111→// Build a March 2025 calendar grid
   112→   112→const CALENDAR_YEAR = 2025
   113→   113→const CALENDAR_MONTH = 2 // March (0-indexed)
   114→   114→const WEEKDAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']
   115→   115→
   116→   116→function buildMarchCalendar() {
   117→   117→  const firstDay = new Date(CALENDAR_YEAR, CALENDAR_MONTH, 1).getDay()
   118→   118→  const daysInMonth = new Date(CALENDAR_YEAR, CALENDAR_MONTH + 1, 0).getDate()
   119→   119→  const cells: (number | null)[] = []
   120→   120→  for (let i = 0; i < firstDay; i++) cells.push(null)
   121→   121→  for (let d = 1; d <= daysInMonth; d++) cells.push(d)
   122→   122→  while (cells.length % 7 !== 0) cells.push(null)
   123→   123→  return cells
   124→   124→}
   125→   125→
   126→   126→function eventsOnDay(day: number | null) {
   127→   127→  if (day === null) return []
   128→   128→  const iso = `${CALENDAR_YEAR}-${String(CALENDAR_MONTH + 1).padStart(
   129→   129→    2,
   130→   130→    '0'
   131→   131→  )}-${String(day).padStart(2, '0')}`
   132→   132→  return EVENTS.filter((e) => e.date === iso)
   133→   133→}
