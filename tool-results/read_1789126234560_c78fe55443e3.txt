    51→function formatDate(iso: string) {
    52→  return new Date(iso).toLocaleDateString('en-US', {
    53→    month: 'short',
    54→    day: 'numeric',
    55→    year: 'numeric',
    56→  })
    57→}

    59→function getCategoryBadgeClass(category: string) {
    60→  switch (category) {
    61→    case 'Campus':
    62→      return 'bg-emerald-100 text-emerald-700'
    63→    case 'Achievement':
    64→      return 'bg-amber-100 text-amber-700'
    65→    case 'Community':
    66→      return 'bg-teal-100 text-teal-700'
    67→    case 'Academics':
    68→      return 'bg-rose-100 text-rose-700'
    69→    case 'Arts':
    70→      return 'bg-orange-100 text-orange-700'
    71→    default:
    72→      return 'bg-muted text-muted-foreground'
    73→  }

    76→// Build a March 2025 calendar grid
    77→const CALENDAR_YEAR = 2025

    79→const WEEKDAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']
    80→
    81→function buildMarchCalendar() {
    82→  const firstDay = new Date(CALENDAR_YEAR, CALENDAR_MONTH, 1).getDay()
    83→  const daysInMonth = new Date(CALENDAR_YEAR, CALENDAR_MONTH + 1, 0).getDate()
    84→  const cells: (number | null)[] = []
    85→  for (let i = 0; i < firstDay; i++) cells.push(null)
    86→  for (let d = 1; d <= daysInMonth; d++) cells.push(d)
    87→  while (cells.length % 7 !== 0) cells.push(null)
    88→  return cells
    89→}
